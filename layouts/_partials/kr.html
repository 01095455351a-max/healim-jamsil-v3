{{/* 한국어 줄바꿈 다듬기. 브라우저는 한국어를 띄어쓰기마다 끊는다. 그래서 "약 1시간 30분~2시간"이 "약 1시간 / 30분~2시간"으로 쪼개지고, "할 수 있습니다"의 '수'가 다음 줄 앞머리에 홀로 남는 일이 생긴다. 읽는 흐름이 끊긴다. 여기서 붙여야 할 자리에 줄바꿈 없는 공백( )을 넣는다. 문구를 고쳐도 같은 규칙이 다시 적용되므로 손으로 손볼 필요가 없다. 쓰는 법 — 본문을 출력하는 자리에서 {{ partial "kr.html" $s.text }} 붙이는 것 1. 시간 표기 약 1시간 30분~2시간 / 40분~1시간 2. 의존명사 + 있다/없다 할 수 있습니다 · 하는 것 없이 3. 한 글자 조사·부사 등 · 뿐은 앞 어절에, 및은 뒤 어절에 붙인다 4. 가운뎃점 신체·정서 (점이 홀로 남지 않게) 5. 고유명사 해아림한의원 잠실점 6. 침/뜸 + 치료 침 치료 · 뜸 치료 */}} {{- $t := . -}} {{- /* 1. 시간 표기 — 숫자와 물결표가 갈라지지 않게 통째로 묶는다 */ -}} {{- $t = $t | replaceRE `(\d+시간)\s+(\d+분)` "${1} ${2}" -}} {{- $t = $t | replaceRE `(약)\s+(\d+[시분])` "${1} ${2}" -}} {{- $t = $t | replaceRE `(\d+[시분]간?)\s*~\s*(\d+[시분])` "${1}⁠~⁠${2}" -}} {{- /* 2. 의존명사가 줄 앞머리에 홀로 남지 않게 앞 어절에 붙인다 */ -}} {{- $t = $t | replaceRE `\s+(수|것|때|점)\s+(있|없)` " ${1} $2" -}} {{- $t = $t | replaceRE `\s+(수|것|때|점|줄|바)(은|는|이|가|을|를|에|도|만|입니다|이다|처럼)` " ${1}$2" -}} {{- /* 3. 한 글자 조사·부사는 앞 어절에 붙인다 */ -}} {{- $t = $t | replaceRE `(\S)\s+(등|뿐)(\s)` "${1} ${2}$3" -}} {{- /* 및은 뒷말과 한 덩어리로 읽힌다 — 앞이 아니라 뒤에 붙인다 */ -}} {{- $t = $t | replaceRE `(\s)및\s+` "${1}및 " -}} {{- /* 4. 가운뎃점(·)에서 줄이 끊기면 점이 다음 줄 앞머리에 홀로 남는다 */ -}} {{- $t = $t | replaceRE `([\p{Hangul}A-Za-z0-9])·([\p{Hangul}A-Za-z0-9])` "${1}⁠·⁠$2" -}} {{- $t = $t | replaceRE `(\S) · (\S)` "${1} · $2" -}} {{- /* 5. 고유명사와 굳어진 표현은 한 덩어리로 */ -}} {{- $t = $t | replaceRE `해아림한의원\s+잠실점` "해아림한의원 잠실점" -}} {{- $t = $t | replaceRE `뿐\s+아니라` "뿐 아니라" -}} {{- /* 6. 치료 이름이 갈라지지 않게 */ -}} {{- $t = $t | replaceRE `(침|뜸|한약)\s+(치료|처방)` "${1} $2" -}} {{- $t | safeHTML -}}
