{{/* 이미지 — assets/의 원본을 화면에 필요한 크기의 WebP로 변환해 내보낸다. 원본(질환 사진 PNG는 장당 2~6MB)을 그대로 내려보내지 않기 위한 것이다. 인자 src assets/ 기준 경로 (예: "design-v3/disease-adhd.png") w 내보낼 가로 픽셀. 화면 표시 폭의 2배를 준다(고해상도 화면 대응). alt 대체 텍스트 style 인라인 style 문자열 (기존 마크업의 값을 그대로 넘긴다) eager 첫 화면에 보이는 이미지면 true — 지연 로딩을 끈다 */}} {{ $r := resources.Get .src }} {{ if $r }} {{ $img := $r }} {{ if lt (int .w) $r.Width }} {{ $img = $r.Resize (printf "%dx webp q80 photo" (int .w)) }} {{ else }} {{ $img = $r.Process "webp q80 photo" }} {{ end }} {{ else }} {{ errorf "image.html: assets/%s 를 찾을 수 없습니다" .src }} {{ end }}
