자필 후기

진료를 받으신 분들이 직접 손으로 적어 주신 체험 후기입니다.

성명은 본인 동의 아래 가려서 게시하며, 연락처 등 신원을 알 수 있는 정보는 담지 않습니다.

{{/* 스캔은 폴더에서 읽는다 — assets/media/reviews/ 에 넣으면 그대로 붙는다. 질환 분류는 data/reviews.yaml 에서 파일명으로 찾는다. 분류표에 없는 파일도 화면에는 나온다(「전체」에서만 보인다) — 파일을 넣을 때마다 표를 고쳐야 하는 구조로 만들지 않기 위해서다. */}} {{ $map := dict }} {{ range site.Data.reviews.items }}{{ $map = merge $map (dict .file (or .group "")) }}{{ end }} {{/* 섞는 순서의 기준. 사이트의 특화 질환 차례를 따르고, 질환 페이지는 없지만 실제로 진료하는 것(야뇨증)을 뒤에 둔다. */}} {{ $order := slice "공황·불안" "틱장애" "ADHD" "강박장애" "우울장애" "불면증" "자율신경실조증" "다한증" "신경성 질환" "야뇨증" }} {{ $scans := slice }} {{ range (resources.Match "media/reviews/*.jpg") }}{{ $scans = $scans | append . }}{{ end }} {{/* 질환이 섞이도록 순서를 다시 짠다. 파일명 순서대로 두면 같은 질환끼리 몰려서, 처음 몇 장만 보고 「이 병원은 이것만 보나」로 읽힌다. 분류별로 줄을 세운 뒤 한 장씩 번갈아 뽑는다 — 파일을 더 넣어도 저절로 다시 섞이고, 분류가 없는 장도 빠지지 않는다. */}} {{ $lanes := slice }} {{ range $order }} {{ $g := . }}{{ $lane := slice }} {{ range $scans }}{{ if eq (index $map (path.Base .Name)) $g }}{{ $lane = $lane | append . }}{{ end }}{{ end }} {{ if $lane }}{{ $lanes = $lanes | append (dict "n" (len $lane) "items" $lane) }}{{ end }} {{ end }} {{ $rest := slice }} {{ range $scans }}{{ $g := index $map (path.Base .Name) }}{{ if not (in $order $g) }}{{ $rest = $rest | append . }}{{ end }}{{ end }} {{ if $rest }}{{ $lanes = $lanes | append (dict "n" (len $rest) "items" $rest) }}{{ end }} {{/* 줄마다 길이가 달라서 한 장씩 번갈아 뽑으면 많은 분류가 뒤에 몰린다. 각 장에 「그 줄에서 몇 번째쯤인가」를 0~1 값으로 매겨 그 순서로 세우면 어느 분류든 처음부터 끝까지 고르게 흩어진다. */}} {{ $keyed := slice }} {{ range $lanes }} {{ $n := .n }} {{ range $i, $r := .items }} {{ $keyed = $keyed | append (dict "k" (div (add (mul $i 1000) 500) $n) "r" $r) }} {{ end }} {{ end }} {{ $mixed := slice }} {{ range sort $keyed "k" }}{{ $mixed = $mixed | append .r }}{{ end }} {{ $scans = $mixed }} {{ if $scans }} {{/* 좌우로 넘겨 보는 띠. 누르면 조금 크게 보이는 창이 열리고, 온전한 크기는 로그인이 필요한 네트워크 홈페이지에서 본다 — 사이트에는 읽을 수 있는 크기의 스캔을 올리지 않는다. */}}
